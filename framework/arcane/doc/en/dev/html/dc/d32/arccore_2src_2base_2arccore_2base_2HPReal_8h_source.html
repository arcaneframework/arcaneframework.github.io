254 static constexpr HPReal accumulate(Real a, HPReal b)
255 {
256 HPReal x(_doTwoSum(a, b.value()));
257 Real c = x.correction() + b.correction();
258 return _doQuickTwoSum(x.value(), c);
259 }
260
261 // Mpi passes through
262 static constexpr HPReal reduce(HPReal a, HPReal b)
263 {
264 HPReal x(_doTwoSum(a.value(), b.value()));
265 Real c = x.correction() + a.correction() + b.correction();
266 return _doQuickTwoSum(x.value(), c);
267 }
268
269 // algo 11 of AC_TWOPRODUCTS Ref 2
270 static constexpr HPReal product(HPReal a, HPReal b)
271 {
272 HPReal x(_doTwoProducts(a.value(), b.value()));
273 Real w = x.correction() + (a.value() * b.correction() + b.value() * a.correction());
274 return HPReal(x.value(), w);
275 }
276 // algo 11 of AC_TWOPRODUCTS Ref 2
277 static constexpr HPReal product(Real a, HPReal b)
278 {
279 HPReal x(_doTwoProducts(a, b.value()));
280 Real w = x.correction() + (a * b.correction());
281 return HPReal(x.value(), w);
282 }
283
284 // algo div2 Ref 6 div2 = a/b
285 // I put parentheses in the evaluation of w to force an order sometimes the compiler options
286 // ATTENTION the last 2 lines of the div2 algorithm are not included (we do not renormalize)
287 // The "mul2" algorithm also contains them and did not include them for "product" because algorithm 11 of Ref 2 does not have them (in my humble opinion)
288 static constexpr HPReal div2(HPReal a, HPReal b)
289 {
290 Real c = a.value() / b.value();
291 HPReal u(_doTwoProducts(c, b.value()));
292 Real w = ((((a.value() - u.value()) + -u.correction()) + a.correction()) - c * b.correction()) / b.value();
293 return HPReal(c, w);
294 }
295 // algo div2 Ref 6 div2 = a/b
296 static constexpr HPReal div2(Real b, HPReal a)
297 {
298 Real c = a.value() / b;
299 HPReal u(_doTwoProducts(c, b));
300 Real w = (((a.value() - u.value()) + -u.correction()) + a.correction()) / b;
301 return HPReal(c, w);
302 }

344 // Correction of ()
345 static constexpr HPReal _doTwoSum(Real a, Real b)
346 {
347 Real value = a + b;
348 Real approx_b = value - a;
349 Real sum_error = (a - (value - approx_b)) + (b - approx_b);
350 return HPReal(value, sum_error);
351 }
352 // correction tests of the values of a and b and we put absolute values
353 static constexpr HPReal _doQuickTwoSum(Real a1, Real b1)
354 {
355 Real a = a1;
356 Real b = b1;
357 Real abs_b1 = (b1 > 0.0) ? b1 : -b1;
358 Real abs_a1 = (a1 > 0.0) ? a1 : -a1;
359 if (abs_b1 > abs_a1) {
360 a = b1;
361 b = a1;
362 }
363 Real value = a + b;
364 Real error_value = (b - (value - a));
365 return HPReal(value, error_value);
366 }
367 // algorithm 4 ref 2 or algorithm 2.4 ref 3 or algorithm 6 p.4
368 static constexpr HPReal _doTwoProducts(Real a, Real b)
369 {
370 Real x = a * b;
371 HPReal aw = SPLIT(a);
372 HPReal bw = SPLIT(b);
373 Real y = aw.correction() * bw.correction() - (((x - aw.value() * bw.value()) - aw.correction() * bw.value()) - aw.value() * bw.correction());
374 return HPReal(x, y);
375 }
376 static constexpr HPReal SPLIT(Real a)
377 {
378 const Real f = 134217729; // 1+ 2^ceil(52/2);
379 Real c = f * a;
380 Real x = c - (c - a);
381 Real y = a - x;
382 return HPReal(x, y);
383 }
