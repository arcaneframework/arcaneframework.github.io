39AlinaCommunicator::
40AlinaCommunicator(IMessagePassingMng* mpm_comm)
41{
42 MessagePassing::Communicator c = mpm_comm->communicator();
43 if (!c.isValid())
44 ARCCORE_FATAL("Invalid 'IMessagePassingMng' communicator. Only MPI implementation is currently supported");
45 m_mpi_communicator = static_cast<MPI_Comm>(c);
46 MPI_Comm_rank(m_mpi_communicator, &rank);

48 m_message_passing_mng = makeRef(mpm_comm);
49};

54void AlinaCommunicator::
55check(bool cond, const String& message)
56{
57 int lc = (cond) ? 1 : 0;
58 int gc = mpAllReduce(m_message_passing_mng.get(), MessagePassing::eReduceType::ReduceMin, cond);
59
60 if (gc != 0)
61 return;
62 IMessagePassingMng* pm = m_message_passing_mng.get();
63 UniqueArray<int> c(size);
64 if (rank == 0)
65 c.resize(size);
66 ConstArrayView<int> in_view(1, &lc);
67 mpGather(pm, in_view, c, 0);
68 if (rank == 0) {
69 std::cerr << "Failed assumption: " << message << std::endl;
70 std::cerr << "Offending processes:";
71 for (int i = 0; i < size; ++i)
72 if (!c[i])
73 std::cerr << " " << i;
74 std::cerr << std::endl;
75 }
76 mpBarrier(pm);
77 ARCCORE_FATAL("CheckError in MessagePassingUtils: {0}", message);
78}
