98void DynamicMeshIncrementalBuilder::
99_printCellFaceInfos(ItemInternal* icell, const String& str)
100{
101 Cell cell(icell);
102 Integer nb_num_back_face = 0;
103 Integer nb_num_front_face = 0;
104 Integer index = 0;
105 for (Face face : cell.faces()) {
106 String add_msg;
107 if (face.backCell() == cell) {
108 ++nb_num_back_face;
109 add_msg = "is a back face";
110 }
111 else {
112 ++nb_num_front_face;
113 add_msg = "is a front face";
114 }
115 info() << str << ": celluid=" << cell.uniqueId()
116 << " faceuid=" << face.uniqueId()
117 << " faceindex=" << index
118 << add_msg;
119 ++index;
120 }
121 info() << str << ": celluid=" << cell.uniqueId()
122 << " nbback=" << nb_num_back_face
123 << " nbfront=" << nb_num_front_face;
124}

136void DynamicMeshIncrementalBuilder::
137addCells(Integer nb_cell, Int64ConstArrayView cells_infos,
138 Int32 owner, Int32ArrayView cells,
139 bool allow_build_face)
140{
141 // With ARCANE_BULK_ADD_CELLS=1, addCellsBulk() builds the same mesh by filling
142 // the family arrays instead of walking item by item. It returns false, having
143 // touched nothing, on a mesh it does not build, and the loop below then does
144 // the work.
145 if (m_use_bulk_add_cells && addCellsBulk(nb_cell, cells_infos, owner, cells, allow_build_face))
146 return;
147 ItemTypeMng* itm = m_item_type_mng;
148
149 Integer cells_infos_index = 0;
150 bool add_to_cells = cells.size() != 0;
151 if (add_to_cells && nb_cell != cells.size())
152 ARCANE_THROW(ArgumentException, "return array 'cells' has to have same size as number of cells");
153 auto* adder = m_one_mesh_item_adder;
154 for (Integer i_cell = 0; i_cell < nb_cell; ++i_cell) {
155 ItemTypeId item_type_id(CheckedConvert::toInt16(cells_infos[cells_infos_index]));
156 ++cells_infos_index;
157 Int64 cell_unique_id = cells_infos[cells_infos_index];
158 ++cells_infos_index;
159
160 Int32 current_cell_nb_node = 0;
161 if (item_type_id == ITI_GenericPolyhedron) {
162 // For poly types, the number of nodes is the first value in the connectivity
163 current_cell_nb_node = static_cast<Int32>(cells_infos[cells_infos_index]);
164 }
165 else {
166 ItemTypeInfo* it = itm->typeFromId(item_type_id);
167 current_cell_nb_node = it->nbLocalNode();
168 }
169 ConstArrayView<Int64> current_cell_nodes_uid(current_cell_nb_node, &cells_infos[cells_infos_index]);
170
171 ItemInternal* cell = adder->addOneCell(item_type_id, cell_unique_id, owner,
172 current_cell_nodes_uid, allow_build_face);
173
174 if (add_to_cells)
175 cells[i_cell] = cell->localId();
176 cells_infos_index += current_cell_nb_node;
177 }
178}

184static const Integer MAX_BULK_FACE_NODE = 32;
185
205bool DynamicMeshIncrementalBuilder::
206addCellsBulk(Integer nb_cell, Int64ConstArrayView cells_infos,
207 Integer sub_domain_id, Int32ArrayView cells,
208 bool allow_build_face)
209{
210 DynamicMesh* mesh = m_mesh;
211 ItemTypeMng* itm = m_item_type_mng;
212 const bool add_to_cells = cells.size() != 0;
213 if (add_to_cells && nb_cell != cells.size())
214 ARCANE_THROW(ArgumentException, "return array 'cells' has to have same size as number of cells");
215
216 // What this path does not know how to build. Anything here falls back.
217 if (nb_cell <= 0 || !allow_build_face)
218 return false;
219 if (mesh->parentMesh())
220 return false;
221 if (!mesh->meshKind().isMonoDimension())
222 return false;
223 const Int32 dimension = mesh->dimension();
224 if (dimension < 1 || dimension > 3)
225 return false;
226 NodeFamily& node_family = mesh->trueNodeFamily();
227 EdgeFamily& edge_family = mesh->trueEdgeFamily();
228 FaceFamily& face_family = mesh->trueFaceFamily();
229 CellFamily& cell_family = mesh->trueCellFamily();
230 // With edges, addCells() always connects a cell to its edges and an edge to
231 // its nodes, and the other edge connectivities only when the mesh keeps them.
232 const bool has_edge = hasEdge();
233 auto has_connectivity = [&](Integer c) { return Connectivity::hasConnectivity(m_connectivity, c); };
234 const bool has_node_edge = has_edge && has_connectivity(Connectivity::CT_NodeToEdge);
235 const bool has_edge_cell = has_edge && has_connectivity(Connectivity::CT_EdgeToCell);
236 const bool has_face_edge = has_edge && has_connectivity(Connectivity::CT_FaceToEdge);
237 const bool has_edge_face = has_edge && has_connectivity(Connectivity::CT_EdgeToFace);
238 if (has_edge) {
239 if (!has_connectivity(Connectivity::CT_CellToEdge) || !has_connectivity(Connectivity::CT_EdgeToNode))
240 return false;
241 if (!edge_family.canAllocateBulk())
242 return false;
243 }
244 // Only a fresh allocation: the localIds handed out are consecutive from 0,
245 // and an item already present would have to be looked up, not created. A
246 // family that was deallocated reports no item but keeps its ItemInternal
247 // slots on a free list, so it is not fresh either.
248 if (!node_family.canAllocateBulk() || !face_family.canAllocateBulk() || !cell_family.canAllocateBulk())
249 return false;
250
251 // ------------------------------------------------------------- pre-scan
252 // Sizes every array of the discovery pass, so that not one of them grows while
253 // it is being filled.
254 UniqueArray<Int64> cell_uids(nb_cell);
255 UniqueArray<Int16> cell_type_ids(nb_cell);
256 UniqueArray<Int32> cell_node_index(nb_cell + 1);
257 UniqueArray<Int32> cell_face_index(nb_cell + 1);
258 UniqueArray<Int32> cell_edge_index(nb_cell + 1);
259 Int32 total_cell_node = 0;
260 Int32 total_cell_face = 0;
261 Int32 total_cell_edge = 0;
262 Int32 total_face_edge = 0;
263 Int32 max_face_node = 0;
264 bool has_high_order_face = false;
265 {
266 Integer k = 0;
267 for (Integer i = 0; i < nb_cell; ++i) {
268 ItemTypeId type_id{ static_cast<Int16>(cells_infos[k]) };
269 // A generic polyhedron carries its own node count in cells_infos and
270 // its faces elsewhere: addCells() builds it.
271 if (type_id == ITI_GenericPolyhedron)
272 return false;
273 ItemTypeInfo* type = itm->typeFromId(type_id);
274 const Integer nb_node = type->nbLocalNode();
275 const Integer nb_face = type->nbLocalFace();
276 cell_type_ids[i] = type_id;
277 cell_uids[i] = cells_infos[k + 1];
278 cell_node_index[i] = total_cell_node;
279 cell_face_index[i] = total_cell_face;
280 cell_edge_index[i] = total_cell_edge;
281 total_cell_node += nb_node;
282 total_cell_face += nb_face;
283 if (has_edge)
284 total_cell_edge += type->nbLocalEdge();
285 for (Integer f = 0; f < nb_face; ++f) {
286 const Integer n = type->localFace(f).nbNode();
287 if (n > MAX_BULK_FACE_NODE)
288 return false; // the per-face scratch below is fixed-size
289 if (n > max_face_node)
290 max_face_node = n;
291 if (has_edge)
292 total_face_edge += type->localFace(f).nbEdge();
293 // A face of order 2 or more is connected to all its nodes, but only its
294 // linear nodes know it (see the node to face fill below).
295 ItemTypeInfo* face_type = itm->typeFromId(ItemTypeId::fromInteger(type->localFace(f).typeId()));
296 if (face_type->linearTypeInfo()->nbLocalNode() != n)
297 has_high_order_face = true;
298 }
299 k += 2 + nb_node;
300 if (k > cells_infos.size())
301 return false;
302 }
303 cell_node_index[nb_cell] = total_cell_node;
304 cell_face_index[nb_cell] = total_cell_face;
305 cell_edge_index[nb_cell] = total_cell_edge;
306 }
307
308 // ---------------------------------------------------------------- phase A
309 // Discovery. One pass over cells_infos, in the order addCells() uses, so that
310 // every localId, every uid and every face orientation come out the same.
311
312 UniqueArray<Int32> cell_nodes(total_cell_node);
313 UniqueArray<Int32> cell_faces(total_cell_face);
314 // A cell face creates at most one face, so these bound the face arrays.
315 UniqueArray<Int64> face_uids(total_cell_face);
316 UniqueArray<Int16> face_type_ids(total_cell_face);
317 UniqueArray<Int32> face_node_index(total_cell_face + 1);
318 UniqueArray<Int32> face_nodes(static_cast<Int64>(total_cell_face) * max_face_node);
319 // With faces of order 2 or more, the linear nodes of each face apart: they
320 // are the first of its sorted nodes.
321 UniqueArray<Int32> face_linear_index(has_high_order_face ? total_cell_face + 1 : 0);
322 UniqueArray<Int32> face_linear_nodes(has_high_order_face ? static_cast<Int64>(total_cell_face) * max_face_node : 0);
323 Int32 nb_face_linear_node = 0;
324 if (has_high_order_face)
325 face_linear_index[0] = 0;
326 UniqueArray<Int32> face_back_cell(total_cell_face);
327 UniqueArray<Int32> face_front_cell(total_cell_face);
328 UniqueArray<Int64> node_uids(total_cell_node);
329 Int32 nb_node = 0;
330 Int32 nb_face = 0;
331 Int32 nb_face_node = 0;
332 face_node_index[0] = 0;
333
334 impl::HashTableMap2<Int64, Int32> node_map(total_cell_node / 4 + 16, true);
335 /*
336 * A face is found by walking the faces already attached to its first sorted
337 * node: they are a handful, and they were created within a few cells of each
338 * other, so the walk stays in cache. This is what findFaceInNode2() does, over
339 * arrays this pass owns.
340 */
341 UniqueArray<Int32> node_first_face(total_cell_node);
342 node_first_face.fill(-1);
343 UniqueArray<Int32> face_chain_next(total_cell_face);
344
345 // The edges, when the mesh has some. An edge is its two nodes, the one of
346 // smaller uniqueId first, and is found among the edges attached to that node,
347 // as findEdgeInNode2() finds it. A cell edge creates at most one edge.
348 UniqueArray<Int32> cell_edges(total_cell_edge);
349 UniqueArray<Int64> edge_uids(total_cell_edge);
350 UniqueArray<Int32> edge_node_index(total_cell_edge + 1);
351 UniqueArray<Int32> edge_nodes(2 * static_cast<Int64>(total_cell_edge));
352 UniqueArray<Int32> node_first_edge(has_edge ? total_cell_node : 0);
353 node_first_edge.fill(-1);
354 UniqueArray<Int32> edge_chain_next(total_cell_edge);
355 // A face's edges are connected when the face is created, so they follow the
356 // face order, as the faces' nodes do.
357 UniqueArray<Int32> face_edge_index(has_edge ? total_cell_face + 1 : 0);
358 UniqueArray<Int32> face_edges(total_face_edge);
359 Int32 nb_edge = 0;
360 Int32 nb_face_edge = 0;
361 if (has_edge)
362 face_edge_index[0] = 0;
363 Int64 next_edge_uid = m_one_mesh_item_adder->nextEdgeUid();
364
365 NodesOfItemReorderer reorderer(itm);
366 SmallArray<Int64, 16> work_face_nodes;
367 FixedArray<Int32, MAX_BULK_FACE_NODE> work_face_lids;
368 FixedArray<Int32, MAX_BULK_FACE_NODE> sorted_lids;
369 Int64 next_face_uid = m_one_mesh_item_adder->nextFaceUid();
370 // As in addCells(), a face's uniqueId is either the next one or a hash of its
371 // sorted node uniqueIds.
372 const bool use_hash = m_one_mesh_item_adder->isUseNodeUniqueIdToGenerateEdgeAndFaceUniqueId();
373 FixedArray<Int64, MAX_BULK_FACE_NODE> sorted_uids;
374
375 Integer infos_index = 0;
376 for (Integer i_cell = 0; i_cell < nb_cell; ++i_cell) {
377 ItemTypeId cell_type_id{ static_cast<Int16>(cells_infos[infos_index]) };
378 ItemTypeInfo* cell_type = itm->typeFromId(cell_type_id);
379 const Integer cell_nb_node = cell_type->nbLocalNode();
380 Int64ConstArrayView cell_nodes_uid(cell_nb_node, &cells_infos[infos_index + 2]);
381 infos_index += 2 + cell_nb_node;
382
383 const Int32 node_base = cell_node_index[i_cell];
384 for (Integer i = 0; i < cell_nb_node; ++i) {
385 auto r = node_map.try_emplace(cell_nodes_uid[i], nb_node);
386 if (r.second)
387 node_uids[nb_node++] = cell_nodes_uid[i];
388 cell_nodes[node_base + i] = r.first->second;
389 }
390
391 // addCells() adds a cell's edges before its faces.
392 const Int32 edge_base = cell_edge_index[i_cell];
393 if (has_edge) {
394 const Integer cell_nb_edge = cell_type->nbLocalEdge();
395 for (Integer i_edge = 0; i_edge < cell_nb_edge; ++i_edge) {
396 const ItemTypeInfo::LocalEdge& le = cell_type->localEdge(i_edge);
397 Int32 first = cell_nodes[node_base + le.beginNode()];
398 Int32 second = cell_nodes[node_base + le.endNode()];
399 if (node_uids[first] > node_uids[second])
400 std::swap(first, second);
401 Int32 edge_index = -1;
402 for (Int32 e = node_first_edge[first]; e >= 0; e = edge_chain_next[e]) {
403 if (edge_nodes[2 * e + 1] == second) {
404 edge_index = e;
405 break;
406 }
407 }
408 if (edge_index < 0) {
409 edge_index = nb_edge;
410 if (use_hash) {
411 FixedArray<Int64, 2> uids;
412 uids[0] = node_uids[first];
413 uids[1] = node_uids[second];
414 edge_uids[nb_edge] = MeshUtils::generateHashUniqueId(uids.view());
415 }
416 else
417 edge_uids[nb_edge] = next_edge_uid++;
418 edge_nodes[2 * nb_edge] = first;
419 edge_nodes[2 * nb_edge + 1] = second;
420 edge_node_index[nb_edge] = 2 * nb_edge;
421 edge_chain_next[nb_edge] = node_first_edge[first];
422 node_first_edge[first] = nb_edge;
423 ++nb_edge;
424 }
425 cell_edges[edge_base + i_edge] = edge_index;
426 }
427 }
428
429 // A face is identified by its sorted node list, exactly as findFaceInNode2()
430 // identifies it.
431 const Integer cell_nb_face = cell_type->nbLocalFace();
432 const Int32 face_base = cell_face_index[i_cell];
433 for (Integer i_face = 0; i_face < cell_nb_face; ++i_face) {
434 const ItemTypeInfo::LocalFace& lf = cell_type->localFace(i_face);
435 const Integer face_nb_node = lf.nbNode();
436 work_face_nodes.resize(face_nb_node);
437 for (Integer z = 0; z < face_nb_node; ++z) {
438 const Int32 local = lf.node(z);
439 work_face_nodes[z] = cell_nodes_uid[local];
440 work_face_lids[z] = cell_nodes[node_base + local];
441 }
442 bool is_reorder = false;
443 if (dimension == 1)
444 is_reorder = reorderer.reorder1D(i_face, work_face_nodes[0]);
445 else
446 is_reorder = reorderer.reorder(ItemTypeId::fromInteger(lf.typeId()), work_face_nodes);
447 ConstArrayView<Int64> sorted_nodes = reorderer.sortedNodes();
448
449 // The sorted nodes are a permutation of the face's own nodes, so their
450 // localIds are found among the handful just gathered, with no trip through
451 // the node map.
452 for (Integer z = 0; z < face_nb_node; ++z) {
453 const Int64 uid = sorted_nodes[z];
454 Int32 lid = NULL_ITEM_LOCAL_ID;
455 for (Integer w = 0; w < face_nb_node; ++w) {
456 if (work_face_nodes[w] == uid) {
457 lid = work_face_lids[w];
458 break;
459 }
460 }
461 sorted_lids[z] = lid;
462 }
463
464 const Int32 first_node = sorted_lids[0];
465 Int32 face_index = -1;
466 for (Int32 f = node_first_face[first_node]; f >= 0; f = face_chain_next[f]) {
467 const Int32 begin = face_node_index[f];
468 if (face_node_index[f + 1] - begin != face_nb_node)
469 continue;
470 bool same = true;
471 for (Integer z = 0; same && z < face_nb_node; ++z)
472 same = (face_nodes[begin + z] == sorted_lids[z]);
473 if (same) {
474 face_index = f;
475 break;
476 }
477 }
478 if (face_index < 0) {
479 face_index = nb_face;
480 if (use_hash) {
481 for (Integer z = 0; z < face_nb_node; ++z)
482 sorted_uids[z] = node_uids[sorted_lids[z]];
483 face_uids[nb_face] = MeshUtils::generateHashUniqueId(SmallSpan<const Int64>(sorted_uids.data(), face_nb_node));
484 }
485 else
486 face_uids[nb_face] = next_face_uid++;
487 face_type_ids[nb_face] = static_cast<Int16>(lf.typeId());
488 face_back_cell[nb_face] = NULL_ITEM_LOCAL_ID;
489 face_front_cell[nb_face] = NULL_ITEM_LOCAL_ID;
490 for (Integer z = 0; z < face_nb_node; ++z)
491 face_nodes[nb_face_node + z] = sorted_lids[z];
492 nb_face_node += face_nb_node;
493 face_node_index[nb_face + 1] = nb_face_node;
494 if (has_high_order_face) {
495 const Integer face_nb_linear_node = itm->typeFromId(ItemTypeId::fromInteger(lf.typeId()))->linearTypeInfo()->nbLocalNode();
496 for (Integer z = 0; z < face_nb_linear_node; ++z)
497 face_linear_nodes[nb_face_linear_node + z] = sorted_lids[z];
498 nb_face_linear_node += face_nb_linear_node;
499 face_linear_index[nb_face + 1] = nb_face_linear_node;
500 }
501 face_chain_next[nb_face] = node_first_face[first_node];
502 node_first_face[first_node] = nb_face;
503 if (has_edge) {
504 const Integer face_nb_edge = lf.nbEdge();
505 for (Integer z = 0; z < face_nb_edge; ++z)
506 face_edges[nb_face_edge + z] = cell_edges[edge_base + lf.edge(z)];
507 nb_face_edge += face_nb_edge;
508 face_edge_index[nb_face + 1] = nb_face_edge;
509 }
510 ++nb_face;
511 }
512 cell_faces[face_base + i_face] = face_index;
513 // is_reorder means this cell is in FRONT of the face.
514 Int32& side = is_reorder ? face_front_cell[face_index] : face_back_cell[face_index];
515 if (side != NULL_ITEM_LOCAL_ID)
516 ARCANE_FATAL("face already has a cell on this side (face index={0})", face_index);
517 side = i_cell;
518 }
519 }
520
521 // ---------------------------------------------------------------- phase B
522 // The items themselves, one batch per family. The discovery order is also the
523 // local id order, so the arrays above index the families directly.
524
525 UniqueArray<Int32> lids(std::max(std::max(nb_node, nb_edge), std::max(nb_face, static_cast<Int32>(nb_cell))));
526 node_family.allocateItemsBulk(node_uids.subConstView(0, nb_node), sub_domain_id,
527 lids.subView(0, nb_node));
528 if (has_edge) {
529 edge_node_index[nb_edge] = 2 * nb_edge;
530 edge_family.allocateItemsBulk(edge_uids.subConstView(0, nb_edge), sub_domain_id,
531 lids.subView(0, nb_edge));
532 for (Int32 i = 0; i < nb_edge; ++i) {
533 if (lids[i] != i)
534 ARCANE_FATAL("unexpected localId {0} for index {1} in family {2}", lids[i], i, edge_family.name());
535 }
536 }
537 _allocateBulkByType(&face_family, face_uids.subConstView(0, nb_face),
538 face_type_ids.subConstView(0, nb_face), sub_domain_id, lids);
539 _allocateBulkByType(&cell_family, cell_uids.constView(), cell_type_ids.constView(),
540 sub_domain_id, lids);
541
542 // ---------------------------------------------------------------- phase C
543 // The connectivities. Each is three arrays; the layout has to be the one the
544 // incremental path produces, since compactConnectivityList() checks it: the
545 // null item holds the first block, and every block starts on a multiple of
546 // preAllocatedSize().
547
548 _fillBulkConnectivity(cell_family.nodeConnectivity(), nb_cell, cell_node_index, cell_nodes);
549 _fillBulkConnectivity(cell_family.faceConnectivity(), nb_cell, cell_face_index, cell_faces);
550 _fillBulkConnectivity(face_family.nodeConnectivity(), nb_face,
551 face_node_index.subConstView(0, nb_face + 1),
552 face_nodes.subConstView(0, nb_face_node));
553 // face -> cell is written by FaceFamily, the class allowed to set the
554 // II_Boundary / II_HasBackCell / II_HasFrontCell flags that go with it.
555 face_family.setBulkBackAndFrontCells(face_back_cell.subConstView(0, nb_face),
556 face_front_cell.subConstView(0, nb_face));
557 _fillBulkReverseConnectivity(node_family.cellConnectivity(), nb_node, nb_cell,
558 cell_node_index, cell_nodes);
559 // As in addCells(), a face is added to its linear nodes only, and only when
560 // the mesh keeps the node to face connectivity (a 1D mesh does not, by default).
561 if (has_connectivity(Connectivity::CT_NodeToFace)) {
562 if (has_high_order_face)
563 _fillBulkReverseConnectivity(node_family.faceConnectivity(), nb_node, nb_face,
564 face_linear_index.subConstView(0, nb_face + 1),
565 face_linear_nodes.subConstView(0, nb_face_linear_node));
566 else
567 _fillBulkReverseConnectivity(node_family.faceConnectivity(), nb_node, nb_face,
568 face_node_index.subConstView(0, nb_face + 1),
569 face_nodes.subConstView(0, nb_face_node));
570 }
571 if (has_edge) {
572 const ConstArrayView<Int32> e_index = edge_node_index.subConstView(0, nb_edge + 1);
573 const ConstArrayView<Int32> e_nodes = edge_nodes.subConstView(0, 2 * nb_edge);
574 const ConstArrayView<Int32> f_index = face_edge_index.subConstView(0, nb_face + 1);
575 const ConstArrayView<Int32> f_edges = face_edges.subConstView(0, nb_face_edge);
576 _fillBulkConnectivity(cell_family.edgeConnectivity(), nb_cell, cell_edge_index, cell_edges);
577 _fillBulkConnectivity(edge_family.nodeConnectivity(), nb_edge, e_index, e_nodes);
578 if (has_node_edge)
579 _fillBulkReverseConnectivity(node_family.edgeConnectivity(), nb_node, nb_edge, e_index, e_nodes);
580 if (has_edge_cell)
581 _fillBulkReverseConnectivity(edge_family.cellConnectivity(), nb_edge, nb_cell, cell_edge_index, cell_edges);
582 if (has_face_edge)
583 _fillBulkConnectivity(face_family.edgeConnectivity(), nb_face, f_index, f_edges);
584 if (has_edge_face)
585 _fillBulkReverseConnectivity(edge_family.faceConnectivity(), nb_edge, nb_face, f_index, f_edges);
586 }
587 if (add_to_cells) {
588 for (Integer i = 0; i < nb_cell; ++i)
589 cells[i] = i;
590 }
591
592 m_one_mesh_item_adder->setNextFaceUid(next_face_uid);
593 m_one_mesh_item_adder->setNextEdgeUid(next_edge_uid);
594 m_one_mesh_item_adder->addBulkMeshInfos(nb_node, nb_edge, nb_face, nb_cell);
595
596 return true;
597}

624 if (lids[i] != i)
625 ARCANE_FATAL("unexpected localId {0} for index {1} in family {2}",
626 lids[i], i, family->name());

637void DynamicMeshIncrementalBuilder::
638_fillBulkConnectivity(IncrementalItemConnectivity* c, Int32 nb_source,
639 ConstArrayView<Int32> offsets, ConstArrayView<Int32> targets)
640{
641 const Int32 prealloc = c->preAllocatedSize();
642 const Int32 head = (prealloc <= 0) ? 1 : prealloc;
643 Int64 total = head;
644 for (Int32 i = 0; i < nb_source; ++i) {
645 const Int32 n = offsets[i + 1] - offsets[i];
646 if (n > 0)
647 total += (prealloc > 0) ? (prealloc * ((n + prealloc - 1) / prealloc)) : n;
648 }
649 Int32ArrayView nb_item, index, list;
650 c->resizeBulkConnectivity(nb_source, total, nb_item, index, list);
651 for (Int32 z = 0; z < head; ++z)
652 list[z] = NULL_ITEM_LOCAL_ID;
653 Int64 pos = head;
654 for (Int32 i = 0; i < nb_source; ++i) {
655 const Int32 begin = offsets[i];
656 const Int32 n = offsets[i + 1] - begin;
657 if (n == 0) {
658 index[i] = 0;
659 nb_item[i] = 0;
660 continue;
661 }
662 index[i] = CheckedConvert::toInt32(pos);
663 nb_item[i] = n;
664 for (Int32 z = 0; z < n; ++z)
665 list[pos + z] = targets[begin + z];
666 const Int32 alloc = (prealloc > 0) ? (prealloc * ((n + prealloc - 1) / prealloc)) : n;
667 for (Int32 z = n; z < alloc; ++z)
668 list[pos + z] = NULL_ITEM_LOCAL_ID;
669 pos += alloc;
670 }
671 c->endBulkConnectivity();
672}

675/*---------------------------------------------------------------------------*/
684void DynamicMeshIncrementalBuilder::
685_fillBulkReverseConnectivity(IncrementalItemConnectivity* c, Int32 nb_target,
686 Int32 nb_source, ConstArrayView<Int32> offsets,
687 ConstArrayView<Int32> targets)
688{
689 const Int32 prealloc = c->preAllocatedSize();
690 const Int32 head = (prealloc <= 0) ? 1 : prealloc;
691 UniqueArray<Int32> count(nb_target);
692 count.fill(0);
693 for (Int32 i = 0; i < nb_source; ++i) {
694 for (Int32 z = offsets[i]; z < offsets[i + 1]; ++z)
695 ++count[targets[z]];
696 }
697
698 UniqueArray<Int32> fill_pos(nb_target);
699 Int64 total = head;
700 for (Int32 t = 0; t < nb_target; ++t) {
701 const Int32 n = count[t];
702 if (n == 0) {
703 fill_pos[t] = 0;
704 continue;
705 }
706 fill_pos[t] = CheckedConvert::toInt32(total);
707 total += (prealloc > 0) ? (prealloc * ((n + prealloc - 1) / prealloc)) : n;
708 }
709 Int32ArrayView nb_item, index, list;
710 c->resizeBulkConnectivity(nb_target, total, nb_item, index, list);
711 list.fill(NULL_ITEM_LOCAL_ID);
712 for (Int32 t = 0; t < nb_target; ++t) {
713 nb_item[t] = count[t];
714 index[t] = fill_pos[t];
715 }
716 for (Int32 i = 0; i < nb_source; ++i) {
717 for (Int32 z = offsets[i]; z < offsets[i + 1]; ++z) {
718 const Int32 t = targets[z];
719 list[fill_pos[t]] = i;
720 ++fill_pos[t];
721 }
722 }
723 c->endBulkConnectivity();
724}

856_fillFaceInfo(Integer& nb_face, Integer nb_cell, Int64Array& faces_infos, Int64ConstArrayView cells_infos, std::map<Int64, Int64SharedArray>& cell_to_face_connectivity_info)
857{
858 faces_infos.reserve(2 * cells_infos.size());
859 Int64UniqueArray work_face_sorted_nodes;
860 Int64UniqueArray work_face_orig_nodes;
861 ItemTypeMng* itm = m_item_type_mng;
862 Integer cells_infos_index = 0;
863 nb_face = 0;

867 Int64 current_cell_unique_id = cells_infos[cells_infos_index++];
868 ItemTypeInfo* it = itm->typeFromId(item_type_id);
869 Integer current_cell_nb_node = it->nbLocalNode();
870 Int64ConstArrayView current_cell_node_uids(current_cell_nb_node, &cells_infos[cells_infos_index]);
871 Integer current_cell_nb_face = it->nbLocalFace();
872 for (Integer face_in_cell_index = 0; face_in_cell_index < current_cell_nb_face; ++face_in_cell_index) {
873 const ItemTypeInfo::LocalFace& current_face = it->localFace(face_in_cell_index);
874 Integer current_face_nb_node = current_face.nbNode();
875 work_face_orig_nodes.resize(current_face_nb_node);
876 work_face_sorted_nodes.resize(current_face_nb_node);
877 Integer i = 0;
878 for (Integer node_in_face_index = 0; node_in_face_index < current_face_nb_node; ++node_in_face_index) {
879 work_face_orig_nodes[i++] = current_cell_node_uids[current_face.node(node_in_face_index)];
880 }
881 mesh_utils::reorderNodesOfFace(work_face_orig_nodes, work_face_sorted_nodes);
882 // get face uid: we shouldn't have to guess, it should be given in arguments...
883 Int64 current_face_uid = _findFaceUniqueId(work_face_sorted_nodes, face_nodes_set);
884 if (current_face_uid == NULL_ITEM_ID) // face not already known add to face info

886 current_face_uid = m_face_uid_pool++;
887 faces_infos.add(current_face.typeId()); // face info type
888 faces_infos.add(current_face_uid); // face info uid
889 faces_infos.addRange(work_face_sorted_nodes); // face info node uids
890 _addFaceNodes(face_nodes_set, work_face_sorted_nodes, current_face_uid);
891 ++nb_face;
892 }
893 cell_to_face_connectivity_info[current_cell_unique_id].add(current_face_uid);
894 }
895 cells_infos_index += current_cell_nb_node;
896 }

979void DynamicMeshIncrementalBuilder::
980_fillCellNewInfoNew(Integer nb_cell, Int64ConstArrayView cells_infos, Int64Array& cell_infos2,
981 const std::map<Int64, Int64SharedArray>& cell_to_face_connectivity_info, const std::map<std::pair<Int64, Int64>, Int64>& edge_uid_map)
982{
983 Integer cell_infos2_size_approx = 1 + 3 * (cells_infos.size() + 2 * nb_cell); // supposes as many faces as nodes as edges...
984 cell_infos2.reserve(cell_infos2_size_approx);
985 ItemTypeMng* itm = m_item_type_mng;
986 Integer cells_infos_index = 0;
987 Integer nb_connected_families = hasEdge() ? 3 : 2;
988 cell_infos2.add(nb_connected_families); // nb_connected_families (node and face +/- edge)
989 for (Integer i_cell = 0; i_cell < nb_cell; ++i_cell) {
990 Integer item_type_id = (Integer)cells_infos[cells_infos_index];
991 cell_infos2.add(cells_infos[cells_infos_index++]); // cell type
992 Int64 current_cell_uid = cells_infos[cells_infos_index];
993 cell_infos2.add(cells_infos[cells_infos_index++]); // cell uid
994 ItemTypeInfo* it = itm->typeFromId(item_type_id);
995 //--- Cell node connectivity
996 cell_infos2.add(mesh()->nodeFamily()->itemKind()); // connected_family_id: node family
997 Integer current_cell_nb_node = it->nbLocalNode();
998 cell_infos2.add(current_cell_nb_node); // nb_connected_nodes
999 Int64ConstArrayView current_cell_node_uids(current_cell_nb_node, &cells_infos[cells_infos_index]);
1000 cell_infos2.addRange(current_cell_node_uids); // node ids
1001 //--- Cell face connectivity
1002 cell_infos2.add(mesh()->faceFamily()->itemKind()); // connected_family_id: face family
1003 Integer current_cell_nb_face = it->nbLocalFace();
1004 cell_infos2.add(current_cell_nb_face); // nb_connected_faces
1005 Int64ArrayView current_cell_faces = cell_to_face_connectivity_info.find(current_cell_uid)->second.view();
1006 if (current_cell_nb_face != current_cell_faces.size())
1007 ARCANE_FATAL("Incoherent number of faces for cell {0}. Expected {1} found {2}",
1008 current_cell_uid, current_cell_nb_face, current_cell_faces.size());
1009 cell_infos2.addRange(current_cell_faces); // face ids
1010 //-- Cell edge connectivity
1011 if (hasEdge()) {
1012 cell_infos2.add(mesh()->edgeFamily()->itemKind()); // connected_family_id: edge family
1013 Integer current_cell_nb_edge = it->nbLocalEdge();
1014 cell_infos2.add(current_cell_nb_edge); // nb_connected_edges
1015 for (int edge_index = 0; edge_index < current_cell_nb_edge; ++edge_index) {
1016 Int64 first_node = current_cell_node_uids[it->localEdge(edge_index).beginNode()];
1017 Int64 second_node = current_cell_node_uids[it->localEdge(edge_index).endNode()];
1018 if (first_node > second_node)
1019 std::swap(first_node, second_node); // edge may be oriented negatively in the cell
1020 auto edge_it = edge_uid_map.find(std::make_pair(first_node, second_node));
1021 if (edge_it == edge_uid_map.end())
1022 ARCANE_FATAL("Do not find edge with nodes {0}-{1} in edge uid map. Exiting",
1023 current_cell_node_uids[it->localEdge(edge_index).beginNode()],
1024 current_cell_node_uids[it->localEdge(edge_index).endNode()]);
1025 cell_infos2.add(edge_it->second);

1216_fillItemRelationInfo(ItemData& source_item_relation_data, const ItemData& target_item_dependencies_data,

1218 bool is_source_item_relation_data_empty)
1219{
1220 ARCANE_ASSERT((source_item_relation_data.nbItems() * 2 == source_item_uids_and_types.size()),
1221 ("source item number incoherent between source_item_relation_data and source_item_uids_and_types. Exiting."));
1222 // Fill an ItemData containing source to target relations (target to source relations are included in target_item_data)
1223 auto& source_relation_info = source_item_relation_data.itemInfos();
1224 const auto& target_dependencies_info = target_item_dependencies_data.itemInfos();
1225 auto source_family = source_item_relation_data.itemFamily();
1226 auto target_family = target_item_dependencies_data.itemFamily();
1227 if (!source_family || !target_family)
1228 return;

1230 Integer nb_families_connected_to_target = CheckedConvert::toInteger(target_dependencies_info[0]);
1231 Integer target_info_index = 1; // 0 is nb_connected_families
1232
1233 // Fill map source to target traversing target_item_dependencies_data
1234 for (; target_info_index < target_dependencies_info.size();) {
1235 target_info_index++; // current target item_type
1236 Int64 target_item_uid = target_dependencies_info[target_info_index++]; // current target item uid
1237 for (Integer family_connected_to_target = 0; family_connected_to_target < nb_families_connected_to_target; ++family_connected_to_target) {
1238 Int64 family_connected_to_target_kind = target_dependencies_info[target_info_index++]; // current target item connected family kind
1239 if (family_connected_to_target_kind != source_family->itemKind()) { //this connection info does not concern source family. Skip
1240 Integer nb_non_read_values = CheckedConvert::toInteger(target_dependencies_info[target_info_index++]); // nb_connected_item on this other family (/= to source family)
1241 target_info_index += nb_non_read_values;
1242 continue;
1243 }
1244 else {
1245 Int32 nb_source_item_connected_to_target_item = CheckedConvert::toInt32(target_dependencies_info[target_info_index++]);
1246 for (Integer source_item_index = 0; source_item_index < nb_source_item_connected_to_target_item; ++source_item_index) {
1247 source_to_target_uids[target_dependencies_info[source_item_index + target_info_index]].add(target_item_uid);
1248 }
1249 target_info_index += nb_source_item_connected_to_target_item;
1250 }
1251 }
1252 }
1253 // Fill or append source_item_relation_data traversing the map source_to_target_connectivity
1254 if (is_source_item_relation_data_empty)
1255 _initEmptyRelationInfo(source_relation_info, source_to_target_uids, source_item_uids_and_types,
1256 target_dependencies_info.size(), target_family);
1257 else
1258 _appendInitializedRelationInfo(source_relation_info, source_to_target_uids, source_item_uids_and_types,
1259 target_dependencies_info.size(), target_family);
1260}

1287void DynamicMeshIncrementalBuilder::
1288_appendInitializedRelationInfo(Int64Array& source_relation_info, std::map<Int64, Int64SharedArray>& source_to_target_uids, Int64ConstArrayView source_item_uids_and_types,
1289 Integer approx_relation_size,
1290 IItemFamily const* target_family)
1291{
1292 // Create a working array to build the consolidated info. This array will be copied in to source_relation_info at the end.
1293 Int64UniqueArray source_relation_info_wrk_copy;
1294 source_relation_info_wrk_copy.reserve(source_relation_info.size() + approx_relation_size);
1295 std::set<Int64> treated_items; // To detect eventual source_items not already present in source_relation_info
1296 Integer source_relation_info_index = 0;
1297 Integer nb_connected_family = CheckedConvert::toInteger(source_relation_info[source_relation_info_index++]);
1298 source_relation_info_wrk_copy.add(nb_connected_family + 1); // adding a new connected family
1299 for (; source_relation_info_index < source_relation_info.size();) {
1300 source_relation_info_wrk_copy.add(source_relation_info[source_relation_info_index++]); // item type
1301 Int64 source_item_uid = source_relation_info[source_relation_info_index++]; // get item uid
1302 source_relation_info_wrk_copy.add(source_item_uid);
1303 treated_items.insert(source_item_uid);
1304 for (Integer connected_family_index = 0; connected_family_index < nb_connected_family; ++connected_family_index) {
1305 source_relation_info_wrk_copy.add(source_relation_info[source_relation_info_index++]); // family kind
1306 Integer nb_connected_elements = CheckedConvert::toInteger(source_relation_info[source_relation_info_index++]); // nb connected elements
1307 source_relation_info_wrk_copy.add(nb_connected_elements);
1308 source_relation_info_wrk_copy.addRange(source_relation_info.subConstView(source_relation_info_index, nb_connected_elements));
1309 source_relation_info_index += nb_connected_elements;
1310 }
1311 // Add new connection info
1312 source_relation_info_wrk_copy.add(target_family->itemKind()); // family kind
1313 const auto& connected_items = source_to_target_uids[source_item_uid]; // get connected items
1314 source_relation_info_wrk_copy.add(connected_items.size()); // nb connected elements
1315 source_relation_info_wrk_copy.addRange(connected_items);
1316 }
1317 // Find eventual source_items not already present in source_relation_info : todo check this can happen
1318 for (Integer source_item_uids_and_types_index = 0; source_item_uids_and_types_index < source_item_uids_and_types.size(); source_item_uids_and_types_index += 2) {
1319 Int64 source_item_uid = source_item_uids_and_types[source_item_uids_and_types_index];
1320 if (treated_items.find(source_item_uid) == treated_items.end()) {
1321 source_relation_info_wrk_copy.add(source_item_uids_and_types[source_item_uids_and_types_index + 1]); // item type
1322 source_relation_info_wrk_copy.add(source_item_uid); // item uid
1323 source_relation_info_wrk_copy.add(target_family->itemKind()); // family kind
1324 const auto& connected_items = source_to_target_uids[source_item_uid]; // get connected items
1325 source_relation_info_wrk_copy.add(connected_items.size()); // nb connected elements
1326 source_relation_info_wrk_copy.addRange(connected_items);
1327 // Padd this item info with family info : we added only one family, add void values for the other nb_connected_family families
1328 for (Integer connected_family_index = 0; connected_family_index < nb_connected_family; ++connected_family_index) {
1329 source_relation_info_wrk_copy.add(0); // family id (does not matter)
1330 source_relation_info_wrk_copy.add(0); // nb_connected_elements for this family
1331 }
1332 }
1333 }
1334 // Copy working array into destination array
1335 source_relation_info.resize(source_relation_info_wrk_copy.size());
1336 std::copy(source_relation_info_wrk_copy.begin(), source_relation_info_wrk_copy.end(), source_relation_info.begin());
1337}
