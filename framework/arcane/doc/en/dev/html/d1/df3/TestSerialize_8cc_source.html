71template <typename DataType>
72class SerializeValue
73: public ISerializeValue
74{
75 using ValueTraitsType = ValueTraits<DataType>;
76
77 public:
78
79 SerializeValue()
80 : m_data_type(ValueTraitsType::dataType())
81 {}
82
83 public:
84
85 void serialize(ISerializer* s) override
86 {
87 Int64 size = m_array_values.size();
88 switch (s->mode()) {
89 case ISerializer::ModeReserve:
90 std::cout << "ReserveArray type=" << m_data_type << " size=" << m_array_values.size() << "\n";
91 s->reserveArray(m_array_values);
92 if (size > 0)
93 s->reserve(m_data_type, 1);
94 break;
95 case ISerializer::ModePut:
96 std::cout << "PutArray type=" << m_data_type << " size=" << m_array_values.size() << "\n";
97 s->putArray(m_array_values);
98 if (size > 0)
99 s->put(m_array_values[0]);
100 break;
101 case ISerializer::ModeGet:
102 s->getArray(m_result_array_values);
103 if (size > 0)
104 ValueTraitsType::getValue(s, m_unique_value);
105 }
106 }
107
108 void checkValid() override
109 {
110 std::cout << "ref_size=" << m_array_values.size()
111 << " result_size=" << m_result_array_values.size() << "\n";
112 ASSERT_EQ(m_array_values, m_result_array_values);
113 ASSERT_EQ(m_array_values[0], m_unique_value);
114 }
115
116 void resizeAndFill(Int32 size)
117 {
118 m_array_values.resize(size);
119 ValueFiller::fillRandom(542, m_array_values.span());
120 }
121
122 public:
123
124 UniqueArray<DataType> m_array_values;
125 UniqueArray<DataType> m_result_array_values;
126 DataType m_unique_value = {};
127 eBasicDataType m_data_type = eBasicDataType::Unknown;
128};

157 if (size > 0)
158 s->reserve(m_ref_string[0]);
159 break;
160 case ISerializer::ModePut:
161 std::cout << "PutArray type=String size=" << m_ref_string.size() << "\n";
162 se.putArray(m_ref_string);
163 if (size > 0)
164 s->put(m_ref_string[0]);
165 break;
166 case ISerializer::ModeGet:
167 se.getArray(m_result_string);
168 if (size > 0)
169 s->get(m_unique_value);
170 }
171 }
172
173 void checkValid() override
174 {
175 std::cout << "ref_size=" << m_ref_string.size()
176 << " result_size=" << m_result_string.size() << "\n";
177 for (Integer i = 0; i < m_ref_string.size(); ++i) {
178 // std::cout << m_ref_string[i] << " == " << m_result_string[i] << std::endl;
179 ASSERT_EQ(m_ref_string[i], m_result_string[i]);
180 }
181 if (!m_ref_string.empty()) {
182 // std::cout << m_ref_string[0] << " == " << m_unique_value << std::endl;
183 ASSERT_EQ(m_ref_string[0], m_unique_value);
184 }
185 }
186
187 public:
188
189 UniqueArray<String> m_ref_string;
190 UniqueArray<String> m_result_string;
191 String m_unique_value;

310class SerializeValueList
311{
312 public:
313
314 ~SerializeValueList()
315 {
316 for (ISerializeValue* v : m_values)
317 delete v;
318 }
319 void doSerialize(ISerializer* s)
320 {
321 for (ISerializeValue* v : m_values)
322 v->serialize(s);
323 }
324 void checkValid()
325 {
326 for (ISerializeValue* v : m_values)
327 v->checkValid();
328 }
329
330 template <typename DataType>
331 void add(Int32 size)
332 {
333 auto* sval = new SerializeValue<DataType>();
334 sval->resizeAndFill(size);
335 m_values.add(sval);
336 }
337 void addString(const String& v)
338 {
339 m_values.add(new StringSerializeValue(v));
340 }
341
342 void addString(Span<String> v)
343 {
344 m_values.add(new StringSerializeValue(v));
345 }
346
347 template <typename DataType>
348 void add(Int32 size1, Int32 size2)
349 {
350 auto* sval = new Serialize2DValue<DataType>();
351 sval->resizeAndFill(size1, size2);
352 m_values.add(sval);
353 }
354
355 template <typename DataType, typename Extents>
356 void add(Arcane::MDIndex<Extents::rank(), typename Extents::ExtentIndexType> size)
357 {
358 auto* sval = new MDSerializeValue<DataType, Extents>();
359 sval->resizeAndFill(size);
360 m_values.add(sval);
361 }
362
363 public:
364
365 UniqueArray<ISerializeValue*> m_values;
366};

371void _doMisc()
372{
373 Ref<ISerializer> serializer_ref = createSerializer();
374 ISerializer* serializer = serializer_ref.get();
375
376 SerializeValueList values;
377
378 values.add<Float16>(12679);
379 values.add<BFloat16>(3212);
380 values.add<Int8>(6357);
381 values.add<Float32>(983);
382 values.add<Int16>(16353);
383 values.add<Real>(29123);
384 values.add<Int32>(16);
385 values.add<Byte>(3289);
386 values.add<Int64>(12932);
387 values.add<Float128>(19328);
388 values.add<Int128>(32422);
389
390 values.addString("Ceci est un test de chaîne de caractères");
391 values.addString(UniqueArray<String>({ "Ceci", "est", "un", "test", "de", "chaîne", "de", "caractères" }));
392
393 values.add<Real>(29, 123);
394 values.add<Int32>(16, 23);
395
396 values.add<Float16, Arcane::MDDim1>(125);
397 values.add<BFloat16, Arcane::MDDim2>({ 51, 21 });
398 values.add<Int8, Arcane::MDDim3>({ 12, 9, 45 });
399 values.add<Float32, Arcane::MDDim4>({ 14, 45, 1, 4 });
400 values.add<Int16, Arcane::MDDim1>(15);
401 values.add<Real, Arcane::MDDim2>({ 84, 31 });
402 values.add<Int32, Arcane::MDDim3>({ 25, 19, 41 });
403 values.add<Byte, Arcane::MDDim4>({ 12, 17, 26, 4 });
404 values.add<Int64, Arcane::MDDim1>(18);
405 values.add<Float128, Arcane::MDDim2>({ 41, 48 });
406
407 serializer->setMode(ISerializer::ModeReserve);
408 values.doSerialize(serializer);
409 serializer->allocateBuffer();
410 serializer->setMode(ISerializer::ModePut);
411 values.doSerialize(serializer);
412 serializer->setMode(ISerializer::ModeGet);
413 values.doSerialize(serializer);
414
415 values.checkValid();
416}
