99// TODO make this constructor private to IMeshMng to avoid
100// possible duplicates for MeshHandles associated with the same mesh name.
101MeshHandle::MeshHandleRef::
102MeshHandleRef(ISubDomain* sd, const String& name)
103: m_mesh_name(name)
104, m_sub_domain(sd)
105, m_is_null(name.null())
106{
107 if (!m_is_null)
108 m_user_data_list = new UserDataList();
109 ARCANE_CHECK_POINTER(sd);
110 m_trace_mng = sd->traceMng();
111 m_mesh_mng = sd->meshMng();
112 m_variable_mng = sd->variableMng();
113 m_on_destroy_observable = new Observable();
114 if (auto v = Convert::Type<Int32>::tryParseFromEnvironment("ARCANE_DO_FATAL_IN_MESHHANDLE", true))
115 m_do_fatal_in_mesh_method = v.value() != 0;
116}

131void MeshHandle::MeshHandleRef::
132_setMesh(IMesh* mesh)
133{
134 m_mesh_ptr = mesh;
135 if (mesh)
136 // We need to keep a reference for the parallel mng because it may have
137 // been creating during mesh building and no longer have reference after.
138 m_parallel_mng_ref = makeRef(mesh->parallelMng());
139 else
140 m_parallel_mng_ref.reset();
141}
142
143/*---------------------------------------------------------------------------*/
144/*---------------------------------------------------------------------------*/
145
146void MeshHandle::MeshHandleRef::
147_destroyMesh()
148{
149 // TODO: protect against multiple calls
150 IMesh* mesh = m_mesh_ptr;
151 if (!mesh)
152 return;
153 m_on_destroy_observable->notifyAllObservers();
154 m_user_data_list->clear();
155 // Be careful not to set to null until the end of this routine because the
156 // users of \a m_user_data might need this MeshHandle.
157 m_mesh_ptr = nullptr;
158 delete mesh;
159}

241IMesh* MeshHandle::
242mesh() const
243{
244 IMesh* m = m_ref->mesh();
245 if (m)
246 return m;
247 // Eventually, make a fatal error if the mesh is null. For reasons of
248 // compatibility with the existing code, we return 'nullptr'.
249 bool do_fatal = m_ref->isDoFatalInMeshMethod();
250 if (do_fatal)
251 ARCANE_FATAL("Invalid call for null mesh. Call MeshHandle::hasMesh() before to make sure mesh is valid");
252 return nullptr;
253}
