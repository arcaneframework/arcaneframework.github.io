33void SerializerExtension::
34reserveArray(Span<const String> values) const
35{
36 // size of each String + size of array.
37 m_serializer->reserveInt64(values.size() + 1);
38 for (const String& elem : values) {
39 m_serializer->reserveSpan(elem.bytes());
40 }
41}
42

44/*---------------------------------------------------------------------------*/
45
46void SerializerExtension::
47putArray(Span<const String> values) const
48{
49 m_serializer->putInt64(values.size());
50 for (const String& elem : values) {
51 m_serializer->putInt64(elem.length());
52 m_serializer->putSpan(elem.bytes());
53 }
54}

59void SerializerExtension::
60getArray(Array<String>& values) const
61{
62 const Int64 size_array = m_serializer->getInt64();
63 values.resize(size_array);
64 for (Int64 i = 0; i < size_array; ++i) {
65 const Int64 len = m_serializer->getInt64();
66 UniqueArray<Byte> bytes(len);
67 m_serializer->getSpan(bytes);
68 values[i] = String(bytes.span());
69 }
70}
