28MPI_Comm MessagePassing::
29toMpiCommunicator(IMessagePassingMng* mpm)
30{
31 MPI_Comm mpi_comm = MPI_COMM_NULL;
32 if (mpm) {
33 Communicator comm = mpm->communicator();
34 if (comm.isValid())
35 mpi_comm = static_cast<MPI_Comm>(comm);
