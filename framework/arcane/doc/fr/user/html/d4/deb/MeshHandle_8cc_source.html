98// TODO rendre ce constructeur privé à IMeshMng pour éviter d'avoir des
99// doublons possibles pour les MeshHandle associés au même nom de maillage.
100MeshHandle::MeshHandleRef::
101MeshHandleRef(ISubDomain* sd,const String& name)
102: m_mesh_name(name)
103, m_sub_domain(sd)
104, m_is_null(name.null())
105{
106 if (!m_is_null)
107 m_user_data_list = new UserDataList();
108 ARCANE_CHECK_POINTER(sd);
109 m_trace_mng = sd->traceMng();
110 m_mesh_mng = sd->meshMng();
111 m_variable_mng = sd->variableMng();
112 m_on_destroy_observable = new Observable();
113 if (auto v = Convert::Type<Int32>::tryParseFromEnvironment("ARCANE_DO_FATAL_IN_MESHHANDLE",true))
114 m_do_fatal_in_mesh_method = v.value()!=0;
115}

130void MeshHandle::MeshHandleRef::
131_setMesh(IMesh* mesh)
132{
133 m_mesh_ptr = mesh;
134 if (mesh)
135 // Il faut conserver une référence sur le parallel mng car il peut avoir
136 // été créé pendant la construction du maillage et ne plus avoir de référence ensuite.
137 m_parallel_mng_ref = makeRef(mesh->parallelMng());
138 else
139 m_parallel_mng_ref.reset();
140}
141
142/*---------------------------------------------------------------------------*/
143/*---------------------------------------------------------------------------*/
144
145void MeshHandle::MeshHandleRef::
146_destroyMesh()
147{
148 // TODO: protéger les appels multiples
149 IMesh* mesh = m_mesh_ptr;
150 if (!mesh)
151 return;
152 m_on_destroy_observable->notifyAllObservers();
153 m_user_data_list->clear();
154 // Attention à ne mettre à nul que à la fin de cette routine car les
155 // utilisateurs de \a m_user_data peuvent avoir besoin de ce MeshHandle.
156 m_mesh_ptr = nullptr;
157 delete mesh;
158}

240IMesh* MeshHandle::
241mesh() const
242{
243 IMesh* m = m_ref->mesh();
244 if (m)
245 return m;
246 // A terme, faire un fatal si le maillage est nul. Pour des raisons de
247 // compatibilité avec l'existant, on retourne 'nullptr'.
248 bool do_fatal = m_ref->isDoFatalInMeshMethod();
249 if (do_fatal)
250 ARCANE_FATAL("Invalid call for null mesh. Call MeshHandle::hasMesh() before to make sure mesh is valid");
251 return nullptr;
252}
