372 iterator erase(const_iterator first, const_iterator last)
373 {
374 Int64 first_to_remove = first - begin();
375 Int64 last_to_remove = last - begin();
376 if (first == last)
377 return begin() + last_to_remove;
378
379 ARCCORE_CHECK_AT(first_to_remove, m_md->size);
380 ARCCORE_CHECK_AT(last_to_remove, 1 + m_md->size);
381 ARCCORE_ASSERT((first <= last), ("last is not after first"));
382
383 // Calcule le nombre d'éléments à supprimer
384 Int64 nb_to_remove = last - first;
385
386 // Décale tous les éléments après 'last' vers la gauche du nombre d'éléments à supprimer
387 for (iterator it = begin() + last_to_remove; it != end(); ++it) {
388 *(it - nb_to_remove) = *it;
389 }
390 // Réduit la taille
391 Int64 new_size = m_md->size - nb_to_remove;
392 resize(new_size);
393 // Retourne l'itérateur suivant le dernier élément supprimé.
394 return begin() + first_to_remove;
395 }
